    <script>
        // 全局变量
        let currentLord = '';
        let stats = { minxin: 0, bingli: 0, tili: 0 };
        let storyStage = 1;
        let isTyping = false;
        const clickSound = document.getElementById('clickSound');
        const typingSpeed = 20; // 降低打字速度，提升阅读体验
        const cursor = document.querySelector('.custom-cursor');
        let particles = [];
        let particleCount = 0;
        const maxParticles = 20; // 限制粒子数量，避免性能问题

        // 初始化：页面加载完成后触发开始页打字机效果
        window.addEventListener('DOMContentLoaded', () => {
            initStartPageText();
            initCustomCursor();
        });

        // 初始化开始页文字并触发打字机
        function initStartPageText() {
            const startText = `五代十国，乱世纷争，中原板荡，江南偏安。
吴越钱氏保境安民，后周郭荣北伐拓土，宋祖匡胤黄袍加身……
《太平年》的故事，就发生在这风云变幻的年代。
你将化身一方主公，在乱世中做出抉择：是保境安民，还是逐鹿中原？是坚守一方，还是纳土归宋？
你的每一个选择，都将决定自己与天下的命运。`;
            typeWriter('start-desc', startText);
        }

        // 初始化自定义鼠标指针和粒子拖尾
        function initCustomCursor() {
            // 鼠标移动跟随
            document.addEventListener('mousemove', (e) => {
                // 更新自定义指针位置
                cursor.style.left = `${e.clientX}px`;
                cursor.style.top = `${e.clientY}px`;
                
                // 创建粒子拖尾
                createParticle(e.clientX, e.clientY);
            });

            // 鼠标点击时指针缩放效果
            document.addEventListener('mousedown', () => {
                cursor.style.transform = 'translate(-50%, -100%) rotate(180deg) scale(0.8)';
            });
            document.addEventListener('mouseup', () => {
                cursor.style.transform = 'translate(-50%, -100%) rotate(180deg) scale(1)';
            });
        }

        // 创建拖尾粒子
        function createParticle(x, y) {
            if (particleCount >= maxParticles) return;
            
            const particle = document.createElement('div');
            particle.className = 'particle';
            // 随机偏移，营造拖尾效果
            const offsetX = (Math.random() - 0.5) * 20;
            const offsetY = (Math.random() - 0.5) * 20;
            particle.style.left = `${x + offsetX}px`;
            particle.style.top = `${y + offsetY}px`;
            
            // 随机大小和透明度
            const size = Math.random() * 4 + 2;
            particle.style.width = `${size}px`;
            particle.style.height = `${size}px`;
            particle.style.opacity = Math.random() * 0.8 + 0.2;
            
            // 随机颜色（与主题色匹配）
            const colors = ['#8b5a2b', '#6d4421', '#a67c52'];
            particle.style.backgroundColor = colors[Math.floor(Math.random() * colors.length)];
            
            document.body.appendChild(particle);
            particles.push(particle);
            particleCount++;
            
            // 粒子动画和销毁
            setTimeout(() => {
                particle.style.opacity = 0;
                particle.style.transform = `translate(${Math.random() * 30 - 15}px, ${Math.random() * 30 - 15}px) scale(0)`;
                
                setTimeout(() => {
                    document.body.removeChild(particle);
                    particles = particles.filter(p => p !== particle);
                    particleCount--;
                }, 600);
            }, 10);
        }

        // 播放点击音效（替换为指定音效）
        function playClickSound() {
            clickSound.currentTime = 0;
            clickSound.play().catch(err => console.log("音效播放失败：", err));
        }

        // 打字机效果（稳定版，支持开始页/结局页）
        function typeWriter(elementId, text, callback = null) {
            if (isTyping) return;
            isTyping = true;
            const element = document.getElementById(elementId);
            if (!element) {
                isTyping = false;
                callback && callback();
                return;
            }
            
            element.innerHTML = '';
            let i = 0;
            const cursor = document.createElement('span');
            cursor.className = 'typing-cursor';
            element.appendChild(cursor);

            function typeChar() {
                try {
                    if (i < text.length) {
                        let char = text.charAt(i);
                        if (char === '\n') {
                            element.innerHTML = element.innerHTML.replace(/<span class="typing-cursor"><\/span>$/, '<br>');
                        } else {
                            element.innerHTML = element.innerHTML.replace(/<span class="typing-cursor"><\/span>$/, char);
                        }
                        element.appendChild(cursor);
                        i++;
                        setTimeout(typeChar, typingSpeed);
                    } else {
                        cursor.remove();
                        isTyping = false;
                        callback && callback();
                    }
                } catch (e) {
                    console.error("打字机错误：", e);
                    cursor.remove();
                    isTyping = false;
                    callback && callback();
                }
            }
            typeChar();
        }

        // 切换页面
        function switchPage(pageId) {
            document.querySelectorAll('.page').forEach(page => page.classList.remove('active'));
            const target = document.getElementById(pageId);
            if (target) {
                target.classList.add('active');
                // 如果是开始页，重新触发打字机
                if (pageId === 'start-page') {
                    initStartPageText();
                }
            }
        }

            currentLord = lordType;
            storyStage = 1;
            isTyping = false;

            // 初始化数值和信息

                document.getElementById('power-logo-img').src = 'wuyue_logo.png';
                document.getElementById('text-logo-container').className = 'text-logo logo-wuyue';
                document.getElementById('text-logo-container').textContent = '吴越';
                
                // 初始剧情

        // 更新数值面板
        function updateStats() {
            // 限制数值在0-100之间
            stats.minxin = Math.max(0, Math.min(100, stats.minxin));
            stats.bingli = Math.max(0, Math.min(100, stats.bingli));
            stats.tili = Math.max(0, Math.min(100, stats.tili));

            // 更新文本
            document.getElementById('minxin-text').textContent = `${stats.minxin}/100`;
            document.getElementById('bingli-text').textContent = `${stats.bingli}/100`;
            document.getElementById('tili-text').textContent = `${stats.tili}/100`;

            // 更新进度条
            document.getElementById('minxin-bar').style.width = `${stats.minxin}%`;
            document.getElementById('bingli-bar').style.width = `${stats.bingli}%`;
            document.getElementById('tili-bar').style.width = `${stats.tili}%`;
        }

        // 更新选项按钮（核心修复：确保每个按钮都有独立绑定）
        function updateOptions(optionTexts) {
            const optionArea = document.getElementById('option-area');
            if (!optionArea) return;
            
            optionArea.innerHTML = '';
            optionTexts.forEach((text, index) => {
                const btn = document.createElement('button');
                btn.className = 'btn option-btn';
                btn.textContent = text;
                // 直接绑定，避免闭包问题
                btn.dataset.optionId = index + 1;
                btn.onclick = function() {
                    playClickSound();
                    chooseOption(parseInt(this.dataset.optionId));
                };
                optionArea.appendChild(btn);
            });
        }

        // 选择选项（核心：补全所有分支逻辑）
        function chooseOption(optionId) {
            // 先检查是否触发结局
            if (checkEnding()) return;

            if (currentLord === 'qian') {
                handleQianOption(optionId);
            } else if (currentLord === 'zhao') {
                handleZhaoOption(optionId);
            } else if (currentLord === 'guo') {
                handleGuoOption(optionId);
            }

            // 更新数值并再次检查结局
            updateStats();
            checkEnding();
        }
